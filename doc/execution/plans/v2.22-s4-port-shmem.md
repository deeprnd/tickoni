# Plan: Port Shared Memory to macOS/Windows

> **For Hermes:** Use subagent-driven-development skill to implement this plan task-by-task.

**Goal:** Port Firedancer's `fd_shmem` shared memory infrastructure to macOS and Windows, eliminating the platform-dependent shmem stubs so Tickoni's architecture (multi-process tiles over shared memory) works identically on all platforms.

**Architecture:** Replace the Linux-hugetlbfs-dependent `fd_shmem` implementation with a cross-platform backend:
- **POSIX (Linux/macOS):** `shm_open()` → `mmap(MAP_SHARED | MAP_ANONYMOUS)` — standard POSIX shared memory, no hugetlbfs required.
- **Windows:** `CreateFileMapping(INVALID_HANDLE_VALUE, ...)` → `MapViewOfFile(...)` — Win32 equivalent of anonymous shared memory.

NUMA binding and huge pages are Linux-specific optimizations that become no-ops on macOS/Windows (already stubbed out). The core create/join/unlink operations become portable.

**Tech Stack:** Zig (as per architecture decision flow — Linux-only constraints must be ported to Zig), POSIX `shm_open`/`mmap`, Win32 `CreateFileMapping`/`MapViewOfFile`.

## Task 1: Design the cross-platform shmem interface in Zig

**Objective:** Define the platform-abstracted shared memory interface in Zig, keeping the Firedancer C API as the surface but implementing it via Zig backends.

**Files:**
- Create: `src/tickoni/cross_shmem.zig`

**Step 1: Define the abstract interface**

```zig
/// Cross-platform shared memory backend for Tickoni.
///
/// Replaces the Firedancer fd_shmem Linux-hugetlbfs implementation
/// with portable backends:
/// - POSIX (Linux/macOS): shm_open + mmap
/// - Windows: CreateFileMapping + MapViewOfFile
///
/// NUMA binding is a no-op on macOS/Windows (already stubbed).
/// Huge pages are a Linux optimization; normal 4KB pages are used everywhere.

pub const PageSz = enum(u32) {
    normal = 4096,
    huge = 1048576,    // 1MB — Linux only, maps to normal on macOS/Windows
    gigantic = 1073741824, // 1GB — Linux only, maps to normal on macOS/Windows
};

pub const JoinMode = enum(u8) {
    read_only = 0,
    read_write = 1,
};

pub const JoinInfo = struct {
    ref_cnt: i64 = 0,
    join: ?*anyopaque = null,
    shmem: ?[]align(PageSz.normal) u8 = null,
    page_sz: u64 = 4096,
    page_cnt: u64 = 1,
    mode: JoinMode = .read_write,
    hash: u32 = 0,
    name: [256]u8 = undefined,
};

pub const CreateResult = struct {
    region: ?[]align(PageSz.normal) u8 = null,
    gaddr: usize = 0,
};

pub const Shmem = struct {
    allocator: std.mem.Allocator,
    regions: std.AutoArrayHashMap([]align(PageSz.normal) u8, JoinInfo),

    pub fn init(allocator: std.mem.Allocator) Shmem {
        return .{
            .allocator = allocator,
            .regions = std.AutoArrayHashMap([]align(PageSz.normal) u8, JoinInfo).init(allocator),
        };
    }

    pub fn deinit(self: *Shmem) void {
        var iter = self.regions.keyIterator();
        while (iter.next()) |key| {
            self.allocator.free(key);
        }
        self.regions.deinit();
    }

    /// Create a named shared memory region.
    /// Returns the backing slice on success, error on failure.
    pub fn create(self: *Shmem, name: []const u8, page_sz: PageSz, page_cnt: u64, cpu_idx: u32) CreateResult {
        const actual_page_sz = switch (page_sz) {
            .normal => PageSz.normal,
            // huge/gigantic are Linux optimizations; fall back to normal on macOS/Windows.
            .huge, .gigantic => PageSz.normal,
        };
        const total_sz = actual_page_sz * page_cnt;

        // Platform-specific: create the shared backing store.
        const backing = platformCreate(name, total_sz) catch |err| {
            return CreateResult{};
        };

        const info = JoinInfo{
            .ref_cnt = 1,
            .page_sz = actual_page_sz,
            .page_cnt = page_cnt,
            .mode = .read_write,
        };
        @memcpy(&info.name, name);
        _ = cpu_idx; // NUMA binding is a no-op on macOS/Windows

        self.regions.put(backing, info) catch return CreateResult{};
        return CreateResult{
            .region = backing,
            .gaddr = @intFromPtr(backing.ptr),
        };
    }

    /// Join an existing named shared memory region.
    pub fn join(self: *Shmem, name: []const u8, mode: JoinMode) CreateResult {
        // Look up by name in our region map.
        var found: ?[]align(PageSz.normal) u8 = null;
        var iter = self.regions.iterator();
        while (iter.next()) |entry| {
            if (std.mem.eql(u8, std.mem.sliceTo(&entry.value_ptr.name, 0), name)) {
                found = entry.key_ptr.*;
                entry.value_ptr.ref_cnt += 1;
                break;
            }
        }
        return if (found) |region| CreateResult{
            .region = region,
            .gaddr = @intFromPtr(region.ptr),
        } else CreateResult{};
    }

    /// Leave a shared memory join.
    pub fn leave(self: *Shmem, region: []align(PageSz.normal) u8) void {
        if (self.regions.getKey(region)) |key| {
            var info = self.regions.getEntry(key).?.value_ptr;
            info.ref_cnt -= 1;
            if (info.ref_cnt == 0) {
                self.regions.remove(key);
                platformDestroy(key);
                self.allocator.free(key);
            }
        }
    }
};

// Platform-specific operations — implemented in platform-specific files.
export fn platformCreate(name: [*]const u8, sz: u64) anyerror![]align(PageSz.normal) u8;
export fn platformDestroy(region: []align(PageSz.normal) u8) void;
```

```zig
/// POSIX backend: shm_open + mmap.
/// Works on Linux and macOS. No hugetlbfs required.
export fn platformCreate(name: [*]const u8, sz: u64) anyerror![]align(4096) u8 {
    var name_buf: [256]u8 = undefined;
    const n = std.mem.copyForwards(u8, &name_buf, name[0..std.mem.len(name)]);
    const cstr = std.mem.sliceTo(&name_buf, 0);

    // Remove leading '/' if present (POSIX requires it).
    const shmem_name = if (cstr[0] == '/') cstr[1..] else cstr;

    const fd = libc_shm_open(shmem_name, libc.O_CREAT | libc.O_RDWR, 0o660) catch |err| return err;
    defer libc_shm_close(fd);

    if (libc_ftruncate(fd, @intCast(sz)) != 0) return error.ShmemFtruncateFailed;

    const ptr = libc_mmap(null, @intCast(sz), libc.PROT_READ | libc.PROT_WRITE, libc.MAP_SHARED, fd, 0) catch |err| return err;
    if (ptr == libc.MAP_FAILED) return error.ShmemMmapFailed;

    // Zero-initialize.
    @memset(ptr[0..sz], 0);

    return ptr[0..sz];
}

export fn platformDestroy(region: []align(4096) u8) void {
    // On POSIX, destroying means unmap + unlink.
    // The region slice owns the mapping; munmap it.
    _ = region;
    // Note: the actual unlink is handled by the platform-specific cleanup.
}
```

Create `src/tickoni/cross_shmem_windows.zig`:
```zig
/// Windows backend: CreateFileMapping + MapViewOfFile.
export fn platformCreate(name: [*]const u8, sz: u64) anyerror![]align(4096) u8 {
    // Convert name to wide string for CreateFileMappingW.
    var wide_buf: [256]std.unicode.Utf16Char = undefined;
    const wide = std.unicode.utf8ToUtf16Le(&wide_buf, name[0..std.mem.len(name)]) catch return error.ShmemNameInvalid;

    const handle = kernel32.CreateFileMappingW(
        kernel32.INVALID_HANDLE_VALUE,
        null,
        kernel32.PAGE_READWRITE,
        @intCast(@divExact(sz, 0x100000000)),
        @intCast(sz),
        &wide,
    );
    if (handle == null) return error.ShmemCreateFileMappingFailed;

    const ptr = kernel32.MapViewOfFile(handle, kernel32.FILE_MAP_ALL_ACCESS, 0, 0, sz);
    if (ptr == null) return error.ShmemMapViewFailed;

    // Zero-initialize.
    @memset(ptr[0..sz], 0);

    return ptr[0..sz];
}

export fn platformDestroy(region: []align(4096) u8) void {
    // Unmap the view.
    kernel32.UnmapViewOfFile(region.ptr);
}
```

const cross_shmem = @import("../cross_shmem.zig");

test "cross-platform shmem: create and join region" {

## Risk Assessment

| Risk | Likelihood | Mitigation |
|---|---|---|
| `shm_open` not available on all macOS versions | Low | macOS 10.7+ has `shm_open` (POSIX.1-2008). Older versions need fallback to `tmpfs`. |
| Windows `CreateFileMapping` requires admin for large regions | Low | Tickoni uses small regions (kilobytes to megabytes, not gigabytes). |
| Performance regression vs hugetlbfs | Medium | Normal pages are slower for very large regions, but Tickoni tiles use small memory. No measurable impact. |
| NUMA binding no-op on macOS/Windows | None | Already stubbed; this is expected behavior. |
| Thread safety of cross-shmem registry | Low | Use `std.Thread.Mutex` for the registry map. All platform operations are serialized through the mutex. |
| Zig `@import` path changes break existing code | Medium | Update all `@import("cross_shmem")` references in shim and tile files. |
