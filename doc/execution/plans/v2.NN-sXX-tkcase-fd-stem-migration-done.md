## What this stub does

tkcase is a pipeline tile (1-in, 2-out). The stub:

1. Consumes fragments from tkdedu input
2. Forwards each event to BOTH outputs (tkpoly and tkdisp) unchanged
3. Tracks consumed/forwarded counters
4. Shuts down when event_count is exhausted

**Explicitly NOT in this stub (deferred):**
- Case scope creation (case_id, workflow, actor tracking)
- Capsule ring buffer (for replay/UC-11)
- Case lifecycle audit emission (RECORD_CASE_CREATION/COMPLETION)
- Event attachment to cases
- Any business logic

## Task 1: Create stub C files

### 1a: Create `src/tickoni/c_abi/shim/tk_case_stem.c`

```c
/* tk_case_stem.c — fd_stem stub for tkcase.
 *
 * Current: forward every event from input to both outputs.
 * Future: case scope, capsule building, lifecycle audit.
 *
 * Shared infrastructure:
 *   stem_helpers.h: STK_GET_CTX, STK_UNPRIV_INIT, stk_link_in_init,
 *                   stk_link_out_init, stk_link_out_publish,
 *                   stk_publish_counter, STK_EVENT_EXHAUSTED,
 *                   stk_cnc_liveness, STK_DEFINE_RUN_TILE_ACCESSORS
 *   stem_diag.h:    stk_diag_out_t, stk_diag_out_init, stk_diag_publish
 */

#define _GNU_SOURCE

#include "firedancer.h"
#include "tickoni/c_abi/shim/stem_helpers.h"
#include "tickoni/c_abi/shim/stem_diag.h"
#include "tickoni/c_abi/topo_run/tk_case_stem.h"

#include "disco/topo/fd_topo.h"
#include "util/fd_util.h"

#include <string.h>
#include <stddef.h>

#define TKCASE_MAX_OUTS 2UL    /* tkpoly + tkdisp */
#define TKCASE_MTU      512UL

/* ─── Context ─── */

typedef struct {
    /* Counters */
    uint64_t consumed;            /* fragments from input */
    uint64_t forwarded;           /* fragments to outputs (sum) */

    /* Event staging */
    int      stage_valid;
    ulong    stage_sz;
    uint8_t  stage[TKCASE_MTU];

    /* Topology binding (stem_helpers.h) */
    stk_link_in_t  in[1];
    stk_link_out_t out[TKCASE_MAX_OUTS];

    /* Diagnostic (stem_diag.h) */
    stk_diag_out_t diag;

    /* Lifecycle */
    void *   cnc;
    uint64_t event_count;
    int      input_exhausted;

    fd_topo_tile_t const *tile;
} tkcase_ctx_t;

#define TKCASE_CTX_ALIGN alignof(tkcase_ctx_t)

/* ─── Stem specialization ─── */

#define STEM_NAME                         tkcase
#define STEM_BURST                        TKCASE_MAX_OUTS
#define STEM_LAZY                         ((long)10e6)
#define STEM_CALLBACK_CONTEXT_TYPE        tkcase_ctx_t
#define STEM_CALLBACK_CONTEXT_ALIGN       TKCASE_CTX_ALIGN
#define STEM_CALLBACK_DURING_HOUSEKEEPING(ctx) stk_cnc_liveness( (ctx)->cnc )
#define STEM_CALLBACK_DURING_FRAG         tkcase_during_frag
#define STEM_CALLBACK_AFTER_FRAG          tkcase_after_frag
#define STEM_CALLBACK_SHOULD_SHUTDOWN(ctx) \
  ( (ctx)->input_exhausted || \
    ( (ctx)->cnc && fd_cnc_signal_query( (fd_cnc_t *)(ctx)->cnc ) == FD_CNC_SIGNAL_HALT ) )

#include "disco/stem/fd_stem.c"

#undef STEM_NAME
#undef STEM_CALLBACK_CONTEXT_TYPE
#undef STEM_CALLBACK_CONTEXT_ALIGN
#undef STEM_CALLBACK_DURING_HOUSEKEEPING
#undef STEM_CALLBACK_DURING_FRAG
#undef STEM_CALLBACK_AFTER_FRAG
#undef STEM_CALLBACK_SHOULD_SHUTDOWN

/* ─── Scratch footprint ─── */

static ulong
tkcase_scratch_align_tile( void ) {
    return TKCASE_CTX_ALIGN;
}

static ulong
tkcase_scratch_footprint_tile( fd_topo_tile_t const * tile ) {
    (void)tile;
    return fd_ulong_align_up( sizeof(tkcase_ctx_t), TKCASE_CTX_ALIGN );
}

ulong
tkcase_scratch_footprint( void ) {
    return tkcase_scratch_footprint_tile( NULL );
}

/* ─── Privileged init ─── */

static void
tkcase_privileged_init( fd_topo_t const * topo, fd_topo_tile_t const * tile ) {
    (void)topo; (void)tile;

    void * scratch = fd_topo_obj_laddr( (fd_topo_t *)topo, tile->tile_obj_id );
    tkcase_ctx_t * ctx = STK_GET_CTX(tkcase_ctx_t, scratch, TKCASE_CTX_ALIGN);
    memset( ctx, 0, sizeof(*ctx) );

    /* Read event_count from environment */
    {
        const char *env = getenv("TKCASE_EVENT_COUNT");
        if (env) {
            ctx->event_count = strtoull(env, NULL, 10);
            FD_LOG_NOTICE(( "tkcase: event_count=%lu", (unsigned long)ctx->event_count ));
        }
    }

    ctx->tile = tile;
    FD_LOG_NOTICE(( "tkcase: privileged init ctx=%p", (void*)ctx ));
}

/* ─── Unprivileged init ─── */

static void
tkcase_unprivileged_init( fd_topo_t const * topo, fd_topo_tile_t const * tile ) {
    (void)topo; (void)tile;

    void * scratch = fd_topo_obj_laddr( (fd_topo_t *)topo, tile->tile_obj_id );
    tkcase_ctx_t * ctx = STK_GET_CTX(tkcase_ctx_t, scratch, TKCASE_CTX_ALIGN);

    STK_UNPRIV_INIT(topo, tile, ctx);
    if (FD_UNLIKELY(!ctx->cnc))
        FD_LOG_ERR(( "tkcase: CNC not found for tile %lu", tile->id ));
    if (FD_UNLIKELY(tile->in_cnt != 1UL))
        FD_LOG_ERR(( "tkcase: expected 1 input, got %lu", tile->in_cnt ));
    if (FD_UNLIKELY(tile->out_cnt < TKCASE_MAX_OUTS))
        FD_LOG_ERR(( "tkcase: expected >=%lu outputs, got %lu", TKCASE_MAX_OUTS, tile->out_cnt ));

    /* Input link */
    stk_link_in_init( &ctx->in[0], topo, tile, 0UL, TKCASE_MTU );

    /* Output links — skip optional diagnostic link if present */
    ulong pipeline_out_cnt = TKCASE_MAX_OUTS;
    ulong diag_o = stk_diag_out_find( topo, tile );
    if (diag_o != ULONG_MAX)
        pipeline_out_cnt = tile->out_cnt - 1UL;
    if (FD_UNLIKELY(pipeline_out_cnt < TKCASE_MAX_OUTS))
        FD_LOG_ERR(( "tkcase: expected >=%lu pipeline outputs, got %lu", TKCASE_MAX_OUTS, pipeline_out_cnt ));

    for (ulong o = 0UL; o < pipeline_out_cnt && o < TKCASE_MAX_OUTS; o++) {
        ulong link_idx = o < diag_o ? o : o + 1UL;
        stk_link_out_init( &ctx->out[o], topo, tile, (int)link_idx, TKCASE_MTU );
    }

    /* Diagnostic (optional — no-op if link absent) */
    stk_diag_out_init( &ctx->diag, topo, tile, pipeline_out_cnt );

    ctx->tile = tile;
    FD_LOG_NOTICE(( "tkcase: unprivileged init tile=%lu diag_o=%lu", tile->id, diag_o ));
}

/* ─── DURING_FRAG: stage ─── */

static void
tkcase_during_frag( tkcase_ctx_t * ctx,
                    ulong          in_idx,
                    ulong          seq,
                    ulong          sig,
                    ulong          chunk,
                    ulong          sz,
                    ulong          ctl ) {
    (void)in_idx; (void)seq; (void)sig; (void)ctl;

    if (!stk_validate_frag(sz, TKCASE_MTU, ctx->in[0].wksp)) return;
    if (FD_UNLIKELY(!stk_link_in_contains(&ctx->in[0], chunk))) return;

    ctx->stage_valid = 0;
    memcpy(ctx->stage, stk_link_in_laddr(&ctx->in[0], chunk), sz);
    ctx->stage_sz = sz;
    ctx->stage_valid = 1;
}

/* ─── AFTER_FRAG: forward ─── */

static void
tkcase_after_frag( tkcase_ctx_t *      ctx,
                   ulong               in_idx,
                   ulong               seq,
                   ulong               sig,
                   ulong               sz,
                   ulong               tsorig,
                   ulong               tspub,
                   fd_stem_context_t * stem ) {
    (void)sz; (void)tsorig; (void)tspub;

    if (!ctx->stage_valid) return;
    ctx->stage_valid = 0;

    ctx->consumed += 1UL;

    /* Forward to both outputs */
    for (ulong o = 0UL; o < TKCASE_MAX_OUTS; o++) {
        stk_link_out_publish( &ctx->out[o], stem, o, sig, ctx->stage, ctx->stage_sz );
    }
    ctx->forwarded += TKCASE_MAX_OUTS;

    /* Counter + diag + exhaustion */
    stk_publish_counter( ctx->cnc, 0UL, ctx->forwarded );
    stk_diag_publish( &ctx->diag, stem, in_idx, seq, ctx->consumed, 0UL, 0UL );
    STK_EVENT_EXHAUSTED(ctx, ctx->event_count, ctx->consumed);
}

/* ─── Run entry ─── */

static void
tkcase_run_entry( fd_topo_t *      topo,
                  fd_topo_tile_t * tile ) {
    void * scratch = fd_topo_obj_laddr(topo, tile->tile_obj_id);
    tkcase_ctx_t * ctx = STK_GET_CTX(tkcase_ctx_t, scratch, TKCASE_CTX_ALIGN);

    ulong cnc_obj_id = tk_topo_find_tile_obj(topo, tile->id, "cnc");
    if (FD_UNLIKELY(cnc_obj_id == ULONG_MAX))
        FD_LOG_ERR(( "tkcase: CNC not found" ));
    ctx->cnc = fd_topo_obj_laddr(topo, cnc_obj_id);

    tkcase_run(topo, tile);
    tile->allow_shutdown = 1;
}

/* ─── Run-tile descriptor ─── */

fd_topo_run_tile_t TK_CASE_RUN = {
    .name                     = TILE_ID_TKCASE,
    .keep_host_networking     = 0,
    .allow_connect            = 0,
    .allow_renameat           = 0,
    .rlimit_file_cnt          = 0,
    .rlimit_address_space     = 0,
    .rlimit_data              = 0,
    .rlimit_nproc             = 0,
    .for_tpool                = 0,
    .max_event_sz             = NULL,
    .populate_allowed_seccomp = NULL,
    .populate_allowed_fds     = NULL,
    .scratch_align            = tkcase_scratch_align_tile,
    .scratch_footprint        = tkcase_scratch_footprint_tile,
    .loose_footprint          = NULL,
    .privileged_init          = tkcase_privileged_init,
    .unprivileged_init        = tkcase_unprivileged_init,
    .run                      = tkcase_run_entry,
    .rlimit_file_cnt_fn       = NULL,
};

STK_DEFINE_RUN_TILE_ACCESSORS(
    tk_get_case,
    TK_CASE_RUN,
    tkcase_scratch_align_tile,
    tkcase_scratch_footprint_tile
)

#define TKCASE_CTX(scratch) STK_GET_CTX(tkcase_ctx_t, scratch, TKCASE_CTX_ALIGN)

ulong tkcase_scratch_consumed( void * scratch ) {
    return TKCASE_CTX(scratch)->consumed;
}

ulong tkcase_forwarded( void * scratch ) {
    return TKCASE_CTX(scratch)->forwarded;
}
```
